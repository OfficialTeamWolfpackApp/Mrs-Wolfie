/* ===================================== MRS WOLFIE ENTER ANIMATION ===================================== */ const enterButton = document.getElementById("enterMrsWolfie"); const intro = document.getElementById("wolfieIntro"); const video = document.getElementById("mrsWolfieOpeningVideo"); const skipButton = document.getElementById("skipWolfieIntro"); let isEntering = false; let hasNavigated = false; const destination = enterButton.href; /* ===================================== OPEN HOMEPAGE ===================================== */ function openWolfieHome() { if (hasNavigated) return; hasNavigated = true; video.pause(); window.location.assign(destination); } /* ===================================== START CINEMATIC INTRO ===================================== */ enterButton.addEventListener("click", function(event) { if ( event.ctrlKey || event.metaKey || event.shiftKey || event.altKey || event.button !== 0 ) { return; } event.preventDefault(); if (isEntering) return; isEntering = true; /* ACCESSIBILITY */ if ( window.matchMedia( "(prefers-reduced-motion: reduce)" ).matches ) { openWolfieHome(); return; } /* SHOW CINEMATIC INTRO */ intro.classList.add("active"); intro.setAttribute( "aria-hidden", "false" ); skipButton.focus(); /* PLAY MRS WOLFIE OPENING VIDEO */ video.currentTime = 0; video.addEventListener( "ended", openWolfieHome, { once: true } ); const playAttempt = video.play(); if (playAttempt) { playAttempt.catch(function(error) { console.warn( "Mrs Wolfie opening video could not play:", error ); openWolfieHome(); }); } /* ===================================== SKIP INTRO ===================================== */ skipButton.addEventListener("click", function() { openWolfieHome(); });
